\section{Algorithms}

\begin{algorithm}[H]
    \SetAlgoLined
    \KwIn{Two numbers $a$ and $b$}
    \KwOut{Sum $s = a + b$}
    
    $s \leftarrow a + b$ \;
    \Return $s$ \;
    
    \caption{Add Two Numbers}
\end{algorithm}

\begin{algorithm}[H]
    \SetAlgoLined

    \Return $s$ \;
        
    \caption{Multiply Two Numbers}
